cmake_minimum_required(VERSION 3.21)
project(EoriniZip VERSION 0.5.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(MSVC)
  add_compile_options(/utf-8 /W4 /EHsc)
endif()
add_library(eorinizip_core native/core.cpp native/process.cpp)
target_include_directories(eorinizip_core PUBLIC native)
if(WIN32)
  enable_language(RC)
  target_compile_definitions(eorinizip_core PUBLIC UNICODE _UNICODE NOMINMAX)
  add_library(EoriniZipModernShell SHARED native/modern_shell.cpp native/modern_shell.def)
  target_compile_definitions(EoriniZipModernShell PRIVATE UNICODE _UNICODE NOMINMAX)
  target_link_libraries(EoriniZipModernShell PRIVATE shell32 ole32 user32 uuid)
  add_library(EoriniZipShell SHARED native/shell_menu.cpp native/shell_menu.def)
  set_target_properties(EoriniZipShell PROPERTIES OUTPUT_NAME EoriniZipShell_v3)
  target_compile_definitions(EoriniZipShell PRIVATE UNICODE _UNICODE NOMINMAX)
  target_link_libraries(EoriniZipShell PRIVATE shell32 ole32 user32 uuid gdi32)
  add_executable(EoriniZip WIN32 native/windows.cpp native/windows.rc)
  target_link_libraries(EoriniZip PRIVATE eorinizip_core comctl32 ole32 shell32 uuid msimg32 advapi32 comdlg32 gdiplus)
  if(MSVC)
    target_link_options(EoriniZip PRIVATE /MANIFEST:NO)
  endif()
elseif(APPLE)
  enable_language(OBJCXX)
  add_executable(EoriniZip MACOSX_BUNDLE native/macos.mm assets/EoriniZip.icns)
  set_source_files_properties(assets/EoriniZip.icns PROPERTIES MACOSX_PACKAGE_LOCATION Resources)
  target_compile_options(EoriniZip PRIVATE -fobjc-arc)
  target_link_libraries(EoriniZip PRIVATE eorinizip_core "-framework Cocoa")
  set_target_properties(EoriniZip PROPERTIES MACOSX_BUNDLE_GUI_IDENTIFIER "io.eorinizip.desktop" MACOSX_BUNDLE_ICON_FILE EoriniZip.icns)
else()
  message(FATAL_ERROR "Desktop UI supports Windows and macOS.")
endif()
enable_testing()
add_executable(core_tests tests/core_tests.cpp)
target_link_libraries(core_tests PRIVATE eorinizip_core)
add_test(NAME core COMMAND core_tests)
if(WIN32)
  add_executable(modern_shell_tests tests/modern_shell_tests.cpp)
  target_compile_definitions(modern_shell_tests PRIVATE UNICODE _UNICODE NOMINMAX)
  target_link_libraries(modern_shell_tests PRIVATE shell32 ole32 user32 uuid)
  add_test(NAME modern_shell COMMAND modern_shell_tests $<TARGET_FILE:EoriniZipModernShell>)
  add_executable(shell_menu_tests tests/shell_menu_tests.cpp)
  target_compile_definitions(shell_menu_tests PRIVATE UNICODE _UNICODE NOMINMAX)
  target_link_libraries(shell_menu_tests PRIVATE shell32 ole32 user32 uuid gdi32)
  add_test(NAME shell_menu COMMAND shell_menu_tests $<TARGET_FILE:EoriniZipShell>)
endif()

add_executable(split_tests tests/split_tests.cpp)
target_link_libraries(split_tests PRIVATE eorinizip_core)
# Native auxiliary codecs, built from pinned source with upstream licenses.
find_program(CARGO_EXECUTABLE cargo REQUIRED)
if(WIN32)
  set(CODEC_RUSTFLAGS "-C target-feature=+crt-static")
endif()
set(CODEC_CARGO_TARGET "${CMAKE_BINARY_DIR}/cargo")
foreach(codec unalz unegg easycrab)
  set(codec_binary "${CODEC_CARGO_TARGET}/release/${codec}${CMAKE_EXECUTABLE_SUFFIX}")
  file(GLOB_RECURSE codec_sources CONFIGURE_DEPENDS "${CMAKE_SOURCE_DIR}/third_party/${codec}/src/*.rs")
  add_custom_command(OUTPUT "${codec_binary}"
    COMMAND ${CMAKE_COMMAND} -E env "CARGO_TARGET_DIR=${CODEC_CARGO_TARGET}" "RUSTFLAGS=${CODEC_RUSTFLAGS}"
      ${CARGO_EXECUTABLE} build --locked --release --manifest-path "${CMAKE_SOURCE_DIR}/third_party/${codec}/Cargo.toml"
    DEPENDS ${codec_sources} "third_party/${codec}/Cargo.toml" "third_party/${codec}/Cargo.lock"
    WORKING_DIRECTORY "${CMAKE_SOURCE_DIR}" VERBATIM)
  add_custom_target(codec_${codec} ALL DEPENDS "${codec_binary}")
  foreach(host EoriniZip core_tests split_tests)
    add_dependencies(${host} codec_${codec})
    add_custom_command(TARGET ${host} POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different
      "${codec_binary}" "$<TARGET_FILE_DIR:${host}>/eorinizip-${codec}${CMAKE_EXECUTABLE_SUFFIX}" VERBATIM)
  endforeach()
endforeach()
add_executable(eorinizip-zpaq third_party/zpaq/zpaq.cpp third_party/zpaq/libzpaq.cpp)
target_compile_definitions(eorinizip-zpaq PRIVATE NOJIT)
if(WIN32)
  target_link_libraries(eorinizip-zpaq PRIVATE advapi32)
else()
  target_compile_definitions(eorinizip-zpaq PRIVATE unix)
  find_package(Threads REQUIRED)
  target_link_libraries(eorinizip-zpaq PRIVATE Threads::Threads)
endif()
foreach(host EoriniZip core_tests split_tests)
  add_dependencies(${host} eorinizip-zpaq)
  add_custom_command(TARGET ${host} POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different
    "$<TARGET_FILE:eorinizip-zpaq>" "$<TARGET_FILE_DIR:${host}>/eorinizip-zpaq${CMAKE_EXECUTABLE_SUFFIX}" VERBATIM)
endforeach()
find_program(SEVENZIP_EXECUTABLE NAMES 7zz 7z HINTS "${CMAKE_SOURCE_DIR}/vendor/7zip")
if(SEVENZIP_EXECUTABLE)
  add_test(NAME archive_integration COMMAND core_tests "${SEVENZIP_EXECUTABLE}" "${CMAKE_SOURCE_DIR}/third_party")
  add_test(NAME split_formats COMMAND split_tests "${SEVENZIP_EXECUTABLE}" "${CMAKE_SOURCE_DIR}/third_party" "${CMAKE_SOURCE_DIR}/tests/fixtures/rar")
else()
  message(STATUS "7-Zip not found: archive integration tests require an external full engine")
endif()
